cmake_minimum_required(VERSION 3.20)
project(cuest_find_probe LANGUAGES CXX)

if(NOT PSI4_SOURCE OR NOT EXPECTED_CUEST_LIBRARY)
  message(FATAL_ERROR "Explicit source and canonical library required")
endif()
list(PREPEND CMAKE_MODULE_PATH "${PSI4_SOURCE}/cmake")
find_package(cuEST REQUIRED)
if(NOT TARGET cuEST::cuEST)
  message(FATAL_ERROR "Headers alone must not count as a usable cuEST package")
endif()
get_target_property(actual cuEST::cuEST IMPORTED_LOCATION)
if(NOT actual STREQUAL EXPECTED_CUEST_LIBRARY)
  message(FATAL_ERROR "cuEST resolved ${actual}, expected ${EXPECTED_CUEST_LIBRARY}")
endif()
message(STATUS "Canonical versioned cuEST target verified: ${actual}")
