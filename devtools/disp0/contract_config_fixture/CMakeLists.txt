cmake_minimum_required(VERSION 3.25)
# Configure/generate fixture ONLY. No compiler detection or native build runs.
project(Einsums LANGUAGES NONE)
set(CMAKE_CXX_COMPILER_LOADED TRUE)
set(CMAKE_CXX_COMPILE_OBJECT "${CMAKE_COMMAND} -E false")
set(CMAKE_CXX_CREATE_SHARED_LIBRARY "${CMAKE_COMMAND} -E false")
set(CMAKE_CXX_CREATE_SHARED_MODULE "${CMAKE_COMMAND} -E false")
set(CMAKE_CXX_OUTPUT_EXTENSION ".o")
foreach(_family cudart cublas cublasLt cusolver cusparse nvJitLink)
  if(FIXTURE_MISSING STREQUAL _family)
    continue()
  endif()
  if(FIXTURE_INTERFACE STREQUAL _family)
    add_library(CUDA::${_family} INTERFACE IMPORTED GLOBAL)
  else()
    add_library(CUDA::${_family} UNKNOWN IMPORTED GLOBAL)
    set(_location "${DISP0_PRIVATE_CUDA}/lib64/lib${_family}.so")
    if(FIXTURE_FOREIGN STREQUAL _family)
      set(_location "${FIXTURE_FOREIGN_PATH}")
    endif()
    set_target_properties(CUDA::${_family} PROPERTIES
      IMPORTED_LOCATION "${_location}"
      INTERFACE_INCLUDE_DIRECTORIES "${DISP0_PRIVATE_CUDA}/include")
  endif()
endforeach()
if(FIXTURE_STATIC_NATIVE)
  add_library(Einsums STATIC fixture.cc)
else()
  add_library(Einsums SHARED fixture.cc)
endif()
set_target_properties(Einsums PROPERTIES LINKER_LANGUAGE CXX)
if(NOT FIXTURE_MISSING_PYTHON)
  add_library(PyEinsums MODULE fixture.cc)
  set_target_properties(PyEinsums PROPERTIES LINKER_LANGUAGE CXX OUTPUT_NAME _core)
endif()
if(NOT FIXTURE_MISSING_STUBS)
  add_custom_target(PyEinsumsStubs COMMAND "${CMAKE_COMMAND}" -E false)
endif()
function(fixture_record)
  get_target_property(_native Einsums INSTALL_RPATH)
  get_target_property(_python PyEinsums INSTALL_RPATH)
  get_target_property(_options PyEinsums LINK_OPTIONS)
  file(WRITE "${CMAKE_BINARY_DIR}/fixture-properties.txt" "${_native}\n${_python}\n${_options}\n")
endfunction()
cmake_language(DEFER CALL fixture_record)
